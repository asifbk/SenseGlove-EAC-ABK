✅ BEST & EASIEST METHOD: “Vertex-Stamping Deformation”

When the drill-bit touches the table, you detect the collision, find vertices near the contact point, push them down, and update the mesh.

This works for:

  • Engraving

  • CNC carving

  • Drilling holes

  • Dents

  • Painting height maps

✅ STEP 2 — Add This Script to the Table (DeformableMesh.cs)

⭐ FULL WORKING CODE (Copy–Paste)

csharp
using System.Collections.Generic;
using UnityEngine;

public class DeformableMesh : MonoBehaviour
{
    public float radius = 0.05f;      // brush size
    public float depth = 0.01f;       // how deep engraving goes
    public Transform drillBit;        // assign drill bit tip transform

    private Mesh deformMesh;
    private Vector3[] originalVerts;
    private Vector3[] deformedVerts;

    void Start()
    {
        deformMesh = GetComponent<MeshFilter>().mesh;
        deformMesh.MarkDynamic();

        originalVerts = deformMesh.vertices;
        deformedVerts = deformMesh.vertices;
    }

    void Update()
    {
        if (drillBit == null) return;

        Vector3 localHitPos = transform.InverseTransformPoint(drillBit.position);

        for (int i = 0; i < deformedVerts.Length; i++)
        {
            float dist = Vector3.Distance(localHitPos, deformedVerts[i]);

            if (dist < radius)
            {
                // Push vertex downward creating engraving
                float falloff = 1f - (dist / radius); 
                float push = depth * falloff;

                deformedVerts[i].y -= push;
            }
        }

        deformMesh.vertices = deformedVerts;
        deformMesh.RecalculateNormals();
        deformMesh.RecalculateBounds();
    }
}
