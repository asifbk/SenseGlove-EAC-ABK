CODE IMPLEMENTATION:
────────────────────────────────────────────────────────────────────────
void UpdateFingerLock()
{
    // Lock at hot threshold (90°)
    if (currentHeat >= hotColorThreshold && !isFingerLocked)
    {
        isFingerLocked = true;
        Debug.Log("🔒 INDEX FINGER LOCKED! Heat reached 90° (hot threshold).");
    }
    // Release below hot threshold
    else if (currentHeat < hotColorThreshold && isFingerLocked)
    {
        isFingerLocked = false;
        Debug.Log("🔓 INDEX FINGER RELEASED! Cooled below 90°.");
    }
    
    // Apply 100% force when locked, 0% when unlocked
    float[] ffb = new float[5];
    ffb[1] = isFingerLocked ? indexFingerLockForce : 0f;
    
    internalGlove.QueueFFBLevels(ffb);
    internalGlove.SendHaptics();
}
